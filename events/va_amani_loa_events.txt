# Akil'zon, der Adler
amani_loa_events.1 = {
	type = country_event

	title = amani_loa_events.1.t
	desc = amani_loa_events.1.d
	flavor = amani_loa_events.1.f

	event_image = {
		video = "africa_animism" # TODO: eigenes Bild (VA-027)
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_fire.dds" # TODO: eigenes Icon (VA-027)

	duration = 3

	trigger = {
		# ausgelöst von va_amani_loa_pulse_effect
	}

	# Essenz binden
	option = {
		name = amani_loa_events.1.a
		ai_chance = {
			base = 50
		}
		va_amani_loa_bind_effect = { LOA = akilzon }
	}

	# Loa ehren
	option = {
		name = amani_loa_events.1.b
		default_option = yes
		ai_chance = {
			base = 50
		}
		va_amani_loa_honor_effect = { LOA = akilzon }
	}
}

# Zorn der Loa: einmalig ab drei Bindungen
amani_loa_events.6 = {
	type = country_event

	title = amani_loa_events.6.t
	desc = amani_loa_events.6.d
	flavor = amani_loa_events.6.f

	event_image = {
		video = "votp_conspiring" # TODO: eigenes Bild (VA-027)
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_skull.dds" # TODO: eigenes Icon (VA-027)

	duration = 3

	trigger = {
		# ausgelöst von va_amani_loa_pulse_effect
	}

	# Malacrass opfern: die Priester sind versöhnt, Zul'jins Hof verliert seinen Hexlord
	option = {
		name = amani_loa_events.6.a
		default_option = yes
		ai_chance = {
			base = 50
		}
		add_modifier = {
			name = va_amani_hexlord_sacrificed
			years = 5
		}
		# Malacrass als Charakter (va_quel_thalas_characters.txt, VA-066)
		random_scope_character = {
			limit = { has_template = va_qt_malacrass }
			kill_character = yes
		}
		ig:ig_devout ?= {
			add_modifier = {
				name = va_amani_priests_honored
				years = 5
			}
		}
	}

	# Weitermachen: die Gläubigen wenden sich ab
	option = {
		name = amani_loa_events.6.b
		ai_chance = {
			base = 50
		}
		add_modifier = {
			name = va_amani_loa_forsaken
			years = 10
		}
	}
}

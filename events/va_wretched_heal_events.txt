# Rückfall: Die Wretched sind 3 Monate im Mana-Entzug (Empfänger: Silvermoon)
wretched_heal_events.3 = {
	type = country_event

	title = wretched_heal_events.3.t
	desc = wretched_heal_events.3.d
	flavor = wretched_heal_events.3.f

	event_image = {
		video = "unspecific_sick_in_hospital" # TODO: eigenes Bild (VA-027)
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_skull.dds" # TODO: eigenes Icon (VA-027)

	duration = 3

	trigger = {
		# ausgelöst von va_silvermoon_wretched_relapse_check_effect
	}

	# Notlieferung
	option = {
		name = wretched_heal_events.3.a
		default_option = yes
		add_treasury = -50000
		ai_chance = {
			base = 0
			modifier = {
				trigger = { scaled_debt <= 0.2 }
				add = 50
			}
		}
	}

	# Rückschlag hinnehmen: 6 Monate Heilung verloren
	option = {
		name = wretched_heal_events.3.b
		custom_tooltip = {
			text = wretched_heal_events.3.b.tt
			je:je_silvermoon_wretched ?= { add_progress = { value = -6 name = va_silvermoon_wretched_bar } }
		}
		ai_chance = {
			base = 25
		}
	}
}

# Nachspiel: Süchtige plündern die Mana-Lager (Empfänger: Land mit "Wretched Addicts", alle 2 Jahre)
wretched_heal_events.4 = {
	type = country_event

	title = wretched_heal_events.4.t
	desc = wretched_heal_events.4.d
	flavor = wretched_heal_events.4.f

	event_image = {
		video = "unspecific_vandalized_storefront" # TODO: eigenes Bild (VA-027)
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_protest.dds" # TODO: eigenes Icon (VA-027)

	duration = 3

	trigger = {
		any_scope_state = { has_modifier = va_wretched_addicts }
	}

	# Hart durchgreifen
	option = {
		name = wretched_heal_events.4.a
		default_option = yes
		add_modifier = {
			name = va_silvermoon_mana_store_crackdown
			years = 1
		}
	}

	# Versorgen: in einem betroffenen State endet die Sucht
	option = {
		name = wretched_heal_events.4.b
		add_treasury = -25000
		random_scope_state = {
			limit = { has_modifier = va_wretched_addicts }
			remove_modifier = va_wretched_addicts
		}
	}
}

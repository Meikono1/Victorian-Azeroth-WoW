namespace = wretched_events

# Silvermoon bietet den Wretched den Mana-Pakt an (Empfänger: Wretched, scope:va_wretched_suitor = Silvermoon)
wretched_events.1 = {
	type = country_event

	title = wretched_events.1.t
	desc = wretched_events.1.d
	flavor = wretched_events.1.f

	event_image = {
		video = "unspecific_ruler_speaking_to_people" # TODO: eigenes Bild (VA-027)
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds" # TODO: eigenes Icon (VA-027)

	duration = 3

	trigger = {
		# ausgelöst von va_silvermoon_wretched_pact_button
	}

	# Annehmen: Protektorat von Silvermoon
	option = {
		name = wretched_events.1.a
		default_option = yes
		ai_chance = {
			base = 50
			modifier = {
				trigger = {
					OR = {
						has_modifier = va_mana_scarcity
						has_modifier = va_mana_withdrawal
					}
				}
				add = 40
			}
			# VA-021 Schritt 3: je gefallenem Anführer +10, solange Thaelis lebt und die Sucht verzweifelt ist −30
			modifier = {
				trigger = { scope:va_wretched_suitor = { has_variable = va_wretched_thaelis_slain } }
				add = 10
			}
			modifier = {
				trigger = { scope:va_wretched_suitor = { has_variable = va_wretched_aldaron_slain } }
				add = 10
			}
			modifier = {
				trigger = { scope:va_wretched_suitor = { has_variable = va_wretched_felendren_slain } }
				add = 10
			}
			modifier = {
				trigger = {
					va_wretched_thaelis_alive_trigger = yes
					var:va_wretched_addiction_level >= 3
				}
				add = -30
			}
		}
		scope:va_wretched_suitor = {
			create_diplomatic_pact = {
				country = ROOT
				type = protectorate
			}
			custom_tooltip = {
				text = va_wretched_threat_down_30_tt
				je:je_silvermoon_wretched ?= { add_progress = { value = -30 name = va_silvermoon_wretched_threat_bar } }
			}
			trigger_event = { id = wretched_events.2 popup = yes }
		}
	}

	# Ablehnen
	option = {
		name = wretched_events.1.b
		ai_chance = {
			base = 50
		}
		scope:va_wretched_suitor = {
			custom_tooltip = {
				text = va_wretched_threat_up_5_tt
				je:je_silvermoon_wretched ?= { add_progress = { value = 5 name = va_silvermoon_wretched_threat_bar } }
			}
			trigger_event = { id = wretched_events.3 popup = yes }
		}
	}
}

	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_skull.dds" # TODO: eigenes Icon (VA-027)

	duration = 3

	trigger = {
		has_journal_entry = je_silvermoon_wretched
	}

	# Patrouillen auf der Straße
	option = {
		name = wretched_events.10.a
		default_option = yes
		add_treasury = -25000

			text = va_wretched_threat_down_10_tt

		}
	}

	# Umweg über das Meer: die Straße nach Sunstrider Isle bleibt unsicher
	option = {
		name = wretched_events.10.b
		every_scope_state = {
			limit = { state_region = s:STATE_SUNSTRIDER_ISLE }
			add_modifier = {
				name = va_silvermoon_unsafe_road
				years = 2
			}
		}
	}
}

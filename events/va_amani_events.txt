# Zul'jins Gesandte in Zul'Mashar (Empfänger: Mossflayer, scope:va_amani_envoy = Zul'Jarra)
amani_events.2 = {
	type = country_event

	title = amani_events.2.t
	desc = amani_events.2.d
	flavor = amani_events.2.f

	event_image = {
		video = "africa_diplomats_negotiating" # TODO: eigenes Bild (VA-027)
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds" # TODO: eigenes Icon (VA-027)

	duration = 3

	trigger = {
		# ausgelöst von va_amani_mossflayer_button
	}

	# Annehmen: Marionette von Zul'Jarra
	option = {
		name = amani_events.2.a
		default_option = yes
		ai_chance = {
			base = 30
			modifier = {
				trigger = { scope:va_amani_envoy = { has_variable = va_amani_shadowpine_joined } }
				add = 20
			}
			modifier = {
				trigger = {
					scope:va_amani_envoy = {
						has_variable = va_amani_loa_decided
						var:va_amani_loa_decided >= 2
					}
				}
				add = 20
			}
		}
		scope:va_amani_envoy = {
			create_diplomatic_pact = {
				country = ROOT
				type = puppet
			}
			trigger_event = { id = amani_events.3 popup = yes }
		}
	}

	# Ablehnen
	option = {
		name = amani_events.2.b
		ai_chance = {
			base = 50
		}
		scope:va_amani_envoy = {
			trigger_event = { id = amani_events.4 popup = yes }
		}
	}
}

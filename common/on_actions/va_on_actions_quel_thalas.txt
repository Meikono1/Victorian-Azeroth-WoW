# Prüfung der Todesschneise je State (VA-019): monatlich und sofort bei Besitzwechsel bzw. neuem State (Eroberung, Annexion).
# Vorbild: Vanilla on_monthly_pulse_state, on_state_owner_change, on_state_created in common/on_actions/00_code_on_actions.txt.
# Eigene Einträge werden über on_actions an die Vanilla-On-Actions angehängt (bestätigt in T-044).
on_monthly_pulse_state = {
	on_actions = {
		va_on_state_dead_scar_check
	}
}

# Deathholme-Buff und Selbstheilung (VA-062)
on_monthly_pulse = {
	on_actions = {
		va_on_monthly_deathholme_scourge
	}
}

va_on_monthly_deathholme_scourge = {
	effect = {
		va_update_deathholme_scourge_effect = yes
	}
}

on_state_owner_change = {
	on_actions = {
		va_on_state_dead_scar_check
	}
}

on_state_created = {
	on_actions = {
		va_on_state_dead_scar_check
	}
}

va_on_state_dead_scar_check = {
	effect = { # prüft selbst, ob der State betroffen ist
		va_update_dead_scar_effect = yes
		# Besitzer eines Schneisen-States bekommt das Heil-JE, sobald Quinine erforscht ist (VA-046); Monatspuls fängt die Forschung ab
		if = {
			limit = {
				va_state_on_healable_dead_scar_trigger = yes # ohne Deathholme (VA-062)
				owner = {
					has_technology_researched = quinine
					NOT = { has_journal_entry = je_va_dead_scar }
				}
			}
			owner = { add_journal_entry = { type = je_va_dead_scar } }
		}
	}
}

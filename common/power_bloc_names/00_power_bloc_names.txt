	trigger = {
	    scope:selected_identity = identity:identity_military_treaty_organization

	    	c:GER ?= this

	    	c:ITA ?= this

	    	count >= 2

	    		c:TUR ?= this

	    		c:GER ?= this
	    		c:ITA ?= this
	    	}
	    }
	}
}

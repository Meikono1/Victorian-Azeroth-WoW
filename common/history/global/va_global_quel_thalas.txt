# Startzustand Quel'Thalas: Mana-JE und Todesschneise auf den States entlang von Arthas' Marsch zum Sonnenbrunnen, Auswahl Kono (VA-019).
# Vorbild: Vanilla common/history/global/00_global.txt (add_modifier auf States) und state_region-Variablen in 00_code_on_actions.txt.
GLOBAL = {
	# Betroffene State-Regionen markieren; die Modifier setzt va_update_dead_scar_effect.
	s:STATE_SUNWELL_PLATEAU = { set_variable = { name = va_dead_scar_region } }
	s:STATE_RUINS_OF_SILVERMOON = { set_variable = { name = va_dead_scar_region } }
	s:STATE_WEST_SANCTUM = { set_variable = { name = va_dead_scar_region } }
	s:STATE_EAST_SANCTUM = { set_variable = { name = va_dead_scar_region } }
	s:STATE_TRANQUILLIEN = { set_variable = { name = va_dead_scar_region } }
	s:STATE_DEATHHOLME = { set_variable = { name = va_dead_scar_region } }

	every_state = {
		limit = {
			state_region = { has_variable = va_dead_scar_region }
		}
		va_update_dead_scar_effect = yes
	}

	# Mana-Abhängigkeit ab Tag 1 sichtbar (VA-019)
	every_country = {
		limit = {
			va_country_mana_dependent_trigger = yes
			NOT = { has_journal_entry = je_mana_hunger } # aktiviert sich über possible oft schon selbst (error.log 03.11., 17×)
		}
		add_journal_entry = { type = je_mana_hunger }
	}

	# Todesschneise heilen: jedes Land mit State auf der Schneise und Quinine (VA-046, Kono 2026-11-01)
	every_country = {
		limit = {
			any_scope_state = { va_state_on_dead_scar_trigger = yes }
			has_technology_researched = quinine
			NOT = { has_journal_entry = je_va_dead_scar } # on_state_created war schneller (error.log 01.11.)
		}
		add_journal_entry = { type = je_va_dead_scar }
	}

	# Silvermoon-Ziele (VA-020)
	c:SILVERMOON ?= {
		add_journal_entry = { type = je_silvermoon_deathholme }
		add_journal_entry = { type = je_silvermoon_zul_aman }
		add_journal_entry = { type = je_silvermoon_corridor } # VA-047
		if = {
			limit = { exists = c:WRETCHED_BLOODELFES }
			add_journal_entry = { type = je_silvermoon_wretched } # VA-021
		}
	}

	# Mana-Sucht der Wretched (VA-021), Stufe 3 von 3, endet mit der Vereinigung
	c:WRETCHED_BLOODELFES ?= {
		va_wretched_init_addiction_effect = yes
	}
}

# Ohne Scope (on_monthly_pulse, Start, nach jeder Heilung). Zähler: Script-Wert va_dead_scar_cursed_regions.
# Vorbild: add_modifier mit multiplier wie Vanilla raid_target (scripted_effects/00_chris_scripted_effects.txt).
va_update_deathholme_scourge_effect = {
	c:DEATHHOLME ?= {
		if = {
			limit = { has_modifier = va_deathholme_scourge_tide }
			remove_modifier = va_deathholme_scourge_tide
		}
		if = {
			limit = {
				va_dead_scar_cursed_regions > 0
				s:STATE_DEATHHOLME = { has_variable = va_dead_scar_region }
			}
			add_modifier = {
				name = va_deathholme_scourge_tide
				multiplier = va_dead_scar_cursed_regions
			}
		}
	}
	if = {
		limit = { s:STATE_DEATHHOLME = { has_variable = va_dead_scar_region } }
		if = {
			limit = { va_dead_scar_cursed_regions = 0 }
			s:STATE_DEATHHOLME = { remove_variable = va_dead_scar_region }
			every_state = {
				limit = { state_region = s:STATE_DEATHHOLME }
				va_update_dead_scar_effect = yes
			}
		}
		else = {
			# Schneise in Deathholme schrumpft mit: 20 Stufen je verdorbener Region
			every_state = {
				limit = {
					state_region = s:STATE_DEATHHOLME
					has_building = building_va_dead_scar
				}
			if = {
				limit = { va_dead_scar_cursed_regions <= 1 }
				va_dead_scar_set_level_effect = { level = 20 }
			}
			else_if = {
				limit = { va_dead_scar_cursed_regions <= 2 }
				va_dead_scar_set_level_effect = { level = 40 }
			}
			else_if = {
				limit = { va_dead_scar_cursed_regions <= 3 }
				va_dead_scar_set_level_effect = { level = 60 }
			}
			else_if = {
				limit = { va_dead_scar_cursed_regions <= 4 }
				va_dead_scar_set_level_effect = { level = 80 }
			}
			}
		}
	}
}

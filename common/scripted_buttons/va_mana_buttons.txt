# Mana-Maßnahmen im je_mana_hunger (VA-019, Kono 2026-11-12): Disenchantment, Rationierung an/aus, Fauna aussaugen je Eversong-Region.
# Vorbild: Vanilla je_haitian_debt_payment_button (scripted_buttons/00_hispaniola_buttons.txt), Hausstil va_silvermoon_wretched_buttons.txt.
# Effekte: va_mana_relief_effects.txt. Die Regions-Schalter sind einmal per Skript aus einer Liste erzeugt (gleicher Aufbau je Region).

# Disenchantment: Werkstatt mit 10 Mana für 1 Jahr, Prestige −10 für 2 Jahre, danach 2 Jahre gesperrt
va_mana_disenchant_button = {
	name = "va_mana_disenchant_button"
	desc = "va_mana_disenchant_button_desc"

	visible = {
		va_country_mana_dependent_trigger = yes
	}

	possible = {
		custom_tooltip = {
			text = va_mana_disenchant_cooldown_tt
			NOT = { has_variable = va_mana_disenchant_cooldown }
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = { has_modifier = va_mana_withdrawal }
			add = 50
		}
		else_if = {
			limit = { has_modifier = va_mana_scarcity }
			add = 20
		}
	}

	effect = {
		va_mana_disenchant_effect = yes
	}
}

# Rationierung aus, danach 1 Jahr gesperrt
va_mana_rationing_stop_button = {
	name = "va_mana_rationing_stop_button"
	desc = "va_mana_rationing_stop_button_desc"

	visible = {
		has_modifier = va_mana_rationing
	}

	possible = {
		always = yes
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				NOT = { has_modifier = va_mana_withdrawal }
				NOT = { has_modifier = va_mana_scarcity }
			}
			add = 30
		}
	}

	effect = {
		va_mana_rationing_stop_effect = yes
	}
}

# Ziel 1: Deathholme vertreiben. Erfüllt, sobald Deathholme nicht mehr existiert.
je_silvermoon_deathholme = {
	icon = "gfx/interface/icons/event_icons/event_skull.dds" # TODO: eigenes Icon (VA-027)

	group = je_group_historical_content

	immediate = {
		s:STATE_DEATHHOLME = { add_claim = ROOT }
		s:STATE_WINDRUNNER_VILLAGE = { add_claim = ROOT }
		s:STATE_TRANQUILLIEN = { add_claim = ROOT }
		s:STATE_EAST_SANCTUM = { add_claim = ROOT }
		s:STATE_SUNCROWN_VILLAGE = { add_claim = ROOT }
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			NOT = { exists = c:DEATHHOLME }
		}
		effect = {
			add_modifier = {
				name = va_silvermoon_ghostlands_avenged
				years = 10
			}
			trigger_event = { id = silvermoon_events.2 popup = yes }
		}
	}

	timeout = 5475 # 15 Jahre

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			add_modifier = {
				name = va_silvermoon_deathholme_endures
				years = 10
			}
		}
	}

	transferable = no
	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = yes
}

# Ziel 2: Zul'Aman zerstören. Erfüllt mit vollem Besitz von Zul'Aman und Tor Watha.

# Ziel 3 (VA-047): Korridor nach Tranquillien. Drei Logistik-Etappen ohne Krieg oder Eroberung im Krieg.
# Etappen, Provinzen und Abfragen: va_silvermoon_corridor_triggers.txt / _effects.txt. Vorbild Schalter + Balken: je_va_dead_scar (VA-046).
je_silvermoon_corridor = {
	icon = "gfx/interface/icons/event_icons/event_map.dds" # TODO: eigenes Icon (VA-027)

	group = je_group_historical_content

	scripted_progress_bar = va_silvermoon_corridor_bar
	scripted_button = va_silvermoon_corridor_start_button

	immediate = {
		set_variable = { name = va_corridor_stage value = 1 }
	}

	on_monthly_pulse = {
		effect = {
			va_silvermoon_corridor_pulse_effect = yes
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			va_silvermoon_corridor_open_trigger = yes
		}
		effect = {
			if = {
				limit = { has_modifier = va_silvermoon_corridor_logistics }
				remove_modifier = va_silvermoon_corridor_logistics
			}
			remove_variable = va_corridor_stage
			add_modifier = {
				name = va_silvermoon_road_to_tranquillien
				years = 10
			}
			trigger_event = { id = silvermoon_events.5 popup = yes }
		}
	}

	timeout = 2920 # 8 Jahre, gedacht ca. 5

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			if = {
				limit = { has_modifier = va_silvermoon_corridor_logistics }
				remove_modifier = va_silvermoon_corridor_logistics
			}
			remove_variable = va_corridor_stage
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_silvermoon_corridor_status_war
				trigger = {
					has_modifier = va_silvermoon_corridor_logistics
					is_at_war = yes
				}
			}
			triggered_desc = {
				desc = je_silvermoon_corridor_status_running
				trigger = { has_modifier = va_silvermoon_corridor_logistics }
			}
			triggered_desc = {
				desc = je_silvermoon_corridor_status_idle
				trigger = { always = yes }
			}
		}
	}

	transferable = no
	weight = 110

	should_be_pinned_by_default_uninvolved_or_context = yes
}

# Ziel 4 (VA-021): Die Wretched. Erst Mana-Pakt (Protektorat), dann Eingliedern (schnell, mit Folgen) oder Heilen (teuer, langsam, ohne Folgen).
# Schalter, Balken, Effekte, Trigger: va_silvermoon_wretched_*.txt, Events: va_wretched_events.txt. Vorbild Union: Vanilla romania.2 (events/romania_events.txt).
je_silvermoon_wretched = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds" # TODO: eigenes Icon (VA-027)

	group = je_group_historical_content

	scripted_progress_bar = va_silvermoon_wretched_threat_bar
	scripted_progress_bar = va_silvermoon_wretched_bar
	scripted_button = va_silvermoon_wretched_pact_button
	scripted_button = va_silvermoon_wretched_annex_button
	scripted_button = va_silvermoon_wretched_heal_button
	scripted_button = va_silvermoon_wretched_bounty_thaelis_button
	scripted_button = va_silvermoon_wretched_bounty_aldaron_button
	scripted_button = va_silvermoon_wretched_bounty_felendren_button

	# Ansprüche auf das Wretched-Land (Krieg gegen das eigene Volk, Phase 2c) und Heimat markieren
	immediate = {
		va_silvermoon_wretched_start_effect = yes
	}

	on_monthly_pulse = {
		effect = {
			va_silvermoon_wretched_pulse_effect = yes
			va_silvermoon_wretched_war_effect = yes
			va_silvermoon_wretched_raid_effect = yes
			va_silvermoon_wretched_bounty_pulse_effect = yes
		}
	}

	# Gesetzt von va_silvermoon_wretched_absorb_effect bzw. _heal_complete_effect
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			has_variable = va_wretched_united
		}
		effect = {
			va_silvermoon_wretched_cleanup_effect = yes
			va_silvermoon_wretched_complete_effect = yes
		}
	}

	# Wretched anders untergegangen (Krieg, Dritte); Folgen nur, wenn Silvermoon ihr Land hält
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = { exists = c:WRETCHED_BLOODELFES }
			NOT = { has_variable = va_wretched_united }
		}
		effect = {
			va_silvermoon_wretched_cleanup_effect = yes
			va_silvermoon_wretched_conquered_effect = yes
		}
	}

	timeout = 7300 # 20 Jahre

	# 20 Jahre ohne Lösung: Wretched bleiben, Silvermoon verliert Ansehen
	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			va_silvermoon_wretched_cleanup_effect = yes
			add_modifier = {
				name = va_silvermoon_wretched_neglected
				years = 10
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_silvermoon_wretched_status_war
				trigger = {
					has_modifier = va_silvermoon_wretched_healing
					is_at_war = yes
				}
			}
			triggered_desc = {
				desc = je_silvermoon_wretched_status_starved
				trigger = {
					has_modifier = va_silvermoon_wretched_healing
					NOT = { va_silvermoon_wretched_mana_supplied_trigger = yes }
				}
			}
			triggered_desc = {
				desc = je_silvermoon_wretched_status_healing
				trigger = { has_modifier = va_silvermoon_wretched_healing }
			}
			triggered_desc = {
				desc = je_silvermoon_wretched_status_protectorate
				trigger = { va_silvermoon_wretched_partner_trigger = yes }
			}
			triggered_desc = {
				desc = je_silvermoon_wretched_status_independent
				trigger = { always = yes }
			}
		}
	}

	transferable = no
	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = yes
}

# Objekte mit Erde-States, -Regionen oder -Tags: possible/happened = always = no. Rest unverändert Vanilla.
# possible - filters out achievements on game start so we dont have to check everything all the time
# happened - checks if achievement happened

achievement_mikasa_es_su_casa = {
	possible = {
		has_dlc_feature = ep2_content
		country_has_primary_culture = cu:japanese
	}
	happened = {
		country_has_primary_culture = cu:japanese
		custom_tooltip = {
			text = mikasa_es_su_casa_tt
			any_scope_treaty = {
				any_scope_article = {
					has_type = ship_transfer
					target_country ?= {
						any_primary_culture = {
							has_discrimination_trait = language_hispanophone
						}
					}
					input_ship ?= { 
						is_ship_type = ship_type:ship_type_pre_dreadnought
					}
				}
			}
		}
	}
}
